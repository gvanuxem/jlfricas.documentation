jlBar!: (JLFloat64Vector, JLFloat64Vector) -> Void

jlBar!(x,y) draws a bar plot of y vs x on the existing graph object. Example (Shifted Gaussian Distribution):

vx := jlRange(-3.0..3.0, 25);
vz := map(x +-> (t := x pretend DoubleFloat; coerce(0.5*exp(-(t-1.0)^2/2.0))@JF64), vx);
jlBar!(vx, vz)

jlWireframe!(xGrid, yGrid, zMat)
jlWireframe: ((DoubleFloat, DoubleFloat) -> DoubleFloat, Segment Float, Segment Float) -> Void

jlWireframe(f, a..b, c..d) draws the wireframe of z = f(x,y) as x ranges from min(a, b) to max(a, b) and y ranges from min(c, d) to max(c, d). Example (Monkey Saddle):

f := (u : DoubleFloat, v : DoubleFloat) : DoubleFloat +-> u^3 - 3.0*u*v^2;
jlWireframe(f, -2.0..2.0, -2.0..2.0)
jlWireframe: ((DoubleFloat, DoubleFloat) -> DoubleFloat, Segment Float, Segment Float, PositiveInteger) -> Void

jlWireframe(f, a..b, c..d, n) draws the wireframe using an n by n grid. Example (2D Mexican Hat Wavelet):

f := (u : DoubleFloat, v : DoubleFloat) : DoubleFloat +-> (1.0 - (u^2 + v^2)/2.0)*exp(-(u^2 + v^2)/2.0);
jlWireframe(f, -3.0..3.0, -3.0..3.0, 30)
jlWireframe: ((DoubleFloat, DoubleFloat) -> DoubleFloat, Segment Float, Segment Float, PositiveInteger, String) -> Void

jlWireframe(f, a..b, c..d, n, title) draws the wireframe with title using an n by n grid. Example (Circular Ripple Waves):

f := (u : DoubleFloat, v : DoubleFloat) : DoubleFloat +-> sin(u^2 + v^2);
jlWireframe(f, -3.0..3.0, -3.0..3.0, 30, "Circular Ripple Waves (30x30)")
jlWireframe: ((DoubleFloat, DoubleFloat) -> DoubleFloat, Segment Float, Segment Float, String) -> Void

jlWireframe(f, a..b, c..d, title) draws the wireframe with title. Example (Hyperbolic Paraboloid):

f := (u : DoubleFloat, v : DoubleFloat) : DoubleFloat +-> u^2 - v^2;
jlWireframe(f, -2.0..2.0, -2.0..2.0, "Hyperbolic Paraboloid")
jlWireframe: (JLFloat64Vector, JLFloat64Vector, JLFloat64Matrix) -> Void

jlWireframe(x,y,mat) draws the wireframe surface of mat. Example:

xGrid := jlRange(-3.0..3.0, 40); yGrid := jlRange(-3.0..3.0, 40);
zMat := qnew(40, 40)$JF64MAT;
jlWireframe(xGrid, yGrid, zMat)
